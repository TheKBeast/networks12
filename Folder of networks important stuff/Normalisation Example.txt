--HOW TO MAKE TABLES IN 3RD NORMAL FORM: 

USE master --create database
GO
IF NOT EXISTS (
   SELECT name
   FROM sys.databases
   WHERE name = N'CourseworkDatabase'
)
CREATE DATABASE CourseworkDatabase
GO

CREATE TABLE Person --original table. multiple items in a row (names, emails). UNF.
(
 UserID NVARCHAR(50) PRIMARY KEY NOT NULL
    CONSTRAINT CK_People_UserID_Valid
        CHECK (
            LEN(TRIM(UserID)) > 0 AND
            UserID NOT LIKE '%[ ]%' AND
            UserID NOT LIKE '%/%' AND
            UserID NOT LIKE '%[()''"&%%!?]%'AND
            UserID NOT LIKE '%[[]%' AND
            UserID NOT LIKE '%]%'
        ),
     LoginID NVARCHAR(50) NOT NULL
        CONSTRAINT CK_People_LoginID_Valid
        CHECK (
            LoginID NOT LIKE '%[ ]%' AND
            LoginID NOT LIKE '%/%' AND
            LoginID NOT LIKE '%[()''"&%%!?]%'AND
            LoginID NOT LIKE '%[[]%' AND
            LoginID NOT LIKE '%]%'
        ),
    Surname NVARCHAR(50),
    Forname NVARCHAR(MAX) NOT NULL, --list
    Title NVARCHAR(MAX),
    Position NVARCHAR(MAX),
   Phone VARCHAR(16)
        CONSTRAINT CK_People_Phone_E164
        CHECK (
           Phone IS NULL OR
        (
            LEFT(Phone, 1) = '+' AND
            SUBSTRING(Phone, 2, LEN(Phone) - 1) NOT LIKE '%[^0-9]%' AND
            LEN(Phone) BETWEEN 2 AND 16
        )
        ),
    Email NVARCHAR(MAX) --list
    CONSTRAINT CK_People_Email_At
    CHECK (
    Email IS NULL OR
        (
            Email LIKE '%_@_%._%' AND
            Email NOT LIKE '% %'
        )
    ),
    UserLocation NVARCHAR(50)
);
GO

---EXPLAIN A LITTLE
Because you can have multiple emails, names, and the same position in multiple boxes, you need to have them separated into different tables. This might be a bit overkill for this task, but if it were being used in more ways it would allow for selection of the "first" forename, and infinite emails.
